# └─────────────────────────────────────────────────────────────────────────┘
#
# ── Step 1: PyTorch (must come before everything else) ────────────────────
#   CUDA 13.x:  pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu130
#   CUDA 12.8:  pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu128
#   CPU only:   pip install torch torchvision torchaudio
#
# ── Step 2: Unsloth (detects torch/CUDA versions at install time) ─────────
#   Auto-detect:          pip install unsloth
#   CUDA 13 + Torch 2.10: pip install "unsloth[cu130-torch2100] @ git+https://github.com/unslothai/unsloth.git"
#   CUDA 12 + Torch 2.6:  pip install "unsloth[cu124-torch260]  @ git+https://github.com/unslothai/unsloth.git"
#   (Full list: https://github.com/unslothai/unsloth)
#
# ── Step 3: llama-cpp-python (GGUF inference for local teachers) ──────────
#   With CUDA (recommended):
#     CMAKE_ARGS="-DGGML_CUDA=on" pip install llama-cpp-python
#   CPU only (slow, fallback):
#     pip install llama-cpp-python
#
# ── Step 4: Flash Attention 2 (optional — fixes FlexAttention OOM) ────────
#   pip install wheel ninja
#   MAX_JOBS=2 pip install flash-attn --no-build-isolation
